\chapter{The decomposition theorem}\label{ch:decomposition}

This chapter transcribes [DKS] Section~3 at lemma level, with one
deliberate strengthening forced by our syntactic bar (see
Remark~\ref{rem:bar-gap}).

\begin{definition}[Right factor]\label{def:right-factor}
  \lean{KRTheory.TransMon.rightFactor}\leanok
  \uses{def:transmon}
  For a submonoid $N \le M$ of a transformation monoid $(X, M)$:
  states $X \mathbin{\dot\cup} N$, monoid $N$, acting on the left
  summand by restriction and on the right summand by right
  multiplication.
\end{definition}

\begin{lemma}[Right factor faithfulness]\label{lem:right-factor-faithful}
  \lean{KRTheory.TransMon.rightFactor_faithful}\leanok
  \uses{def:right-factor,def:faithful}
  $(X \mathbin{\dot\cup} N,\, N)$ is faithful.
\end{lemma}
\begin{proof}
  \leanok
  Evaluate at the regular point $1 \in N$.
\end{proof}

\begin{lemma}[Nonempty states]\label{lem:nonempty-of-nonunit}
  \lean{KRTheory.TransMon.nonempty_of_not_isUnit}\leanok
  \uses{def:faithful}
  A faithful transformation monoid whose monoid has a non-unit has a
  nonempty state set.
\end{lemma}
\begin{proof}
  \leanok
  With no states, faithfulness makes every element equal to $1$,
  contradicting the existence of a non-unit.
\end{proof}

\begin{lemma}[Bar generation]\label{lem:bar-generated}
  \lean{KRTheory.TransMon.barMonoid_closure}\leanok
  \uses{def:bar}
  If $M = \langle s \rangle$ then
  $\bar M = \langle \{\,\mathrm{of}\ a \mid a \in s\,\} \cup
  \{\,\mathrm{reset}\ x \mid x \in X\,\}\rangle$.
\end{lemma}
\begin{proof}
  \leanok
  $\mathrm{of}$ is a homomorphism, so $\mathrm{of}$-images of products
  decompose, and resets are generators themselves.
\end{proof}

\begin{remark}[The bar gap]\label{rem:bar-gap}
  \uses{def:bar}
  In [DKS], $\bar M$ adjoins constant \emph{functions}, so
  $(X, \bar M)$ is faithful by construction and their
  Proposition~2.3.2 extracts the division homomorphism from
  cover-uniqueness. Our $\bar M$ (milestone 4) is syntactic:
  $\mathrm{of}\ m \neq \mathrm{reset}\ x_0$ even when $m$ acts as the
  constant map to $x_0$, so our bar need not be faithful and
  cover-uniqueness fails exactly on such cross-constructor pairs.
  The repair is Lemma~\ref{lem:decomp-tag}: a syntactic tag on the
  wreath side separating the two constructor classes, bought by one
  free choice [DKS] leave open.
\end{remark}

\begin{definition}[State map and generator covers]\label{def:decomp-covers}
  \lean{KRTheory.TransMon.decompState, KRTheory.TransMon.coverN,
        KRTheory.TransMon.coverC, KRTheory.TransMon.coverReset}\leanok
  \uses{def:right-factor,def:localdiv-tm,def:bar,def:wreath}
  Write $W = \overline{(Xc, M_c)} \wr
  \overline{(X \mathbin{\dot\cup} N, N)}$. The state map:
  $\varphi(p, \mathrm{inl}\ x) = x$ and
  $\varphi(p, \mathrm{inr}\ n) = p \cdot n$. The covers:
  \[
  \widehat{\mathrm{of}\ n} = (\mathrm{const}\ 1,\ \mathrm{of}\ n),
  \qquad
  \widehat{\mathrm{of}\ c} = (f_c,\ \mathrm{reset}(\mathrm{inr}\ 1)),
  \qquad
  \widehat{\mathrm{reset}\ x} =
    (\mathrm{const}(\mathrm{reset}\ \langle x c\rangle),\
     \mathrm{reset}(\mathrm{inl}\ x)),
  \]
  where $f_c(\mathrm{inl}\ x) = \mathrm{reset}\,\langle x c\rangle$ and
  $f_c(\mathrm{inr}\ n) = \mathrm{of}\,\langle c n c\rangle$ (a local
  divisor element: $cnc \in cM \cap Mc$ definitionally). The front of
  $\widehat{\mathrm{reset}\ x}$ is arbitrary in [DKS]; we choose a
  reset — the tag of Lemma~\ref{lem:decomp-tag}. Cover verification
  ([DKS] §3, transcribed): for $\widehat{\mathrm{of}\ n}$,
  $\varphi((p,\mathrm{inl}\,x)\cdot\widehat{n}) = x\cdot n$ and
  $\varphi((p,\mathrm{inr}\,m)\cdot\widehat{n}) = p\cdot(mn) =
  (p\cdot m)\cdot n$; for $\widehat{\mathrm{of}\ c}$,
  $\varphi((p,\mathrm{inl}\,x)\cdot\widehat{c}) =
  \varphi(x c, \mathrm{inr}\,1) = x\cdot c$ and
  $\varphi((p,\mathrm{inr}\,n)\cdot\widehat{c}) =
  \varphi(p \circ cnc, \mathrm{inr}\,1) = p\cdot nc$; for
  $\widehat{\mathrm{reset}\ x}$, both cases land on $x$.
\end{definition}

\begin{lemma}[The tag]\label{lem:decomp-tag}
  \lean{KRTheory.TransMon.decompInv_of_mem}\leanok
  \uses{def:decomp-covers}
  Every element $w$ of the submonoid generated by the covers satisfies
  \emph{ShapeOK}: either (tag true) the front of $w$ is
  $\mathrm{of}$-shaped at every $\mathrm{inr}$-state and the back of
  $w$ is not an $\mathrm{inl}$-reset, or (tag false) the front of $w$
  is reset-shaped at every state.
\end{lemma}
\begin{proof}
  \leanok
  Generators: the $n$- and $c$-covers are tag-true, the reset covers
  tag-false. Products: tags multiply like the of/reset constructor
  classes of $\bar M$ (true·true = true, anything·false = false,
  false·true = false), by the absorption laws of resets and because
  tag-true backs keep evaluation states in the $\mathrm{inr}$ summand.
\end{proof}

\begin{lemma}[Tagged cover uniqueness]\label{lem:decomp-cover-unique}
  \lean{KRTheory.TransMon.coversAt_unique}\leanok
  \uses{def:decomp-covers,def:faithful,lem:nonempty-of-nonunit}
  If $w$ covers $s$ and $s'$ over $\varphi$ and $s, s'$ have the same
  constructor class, then $s = s'$.
\end{lemma}
\begin{proof}
  \leanok
  $\varphi$ is onto, so $s$ and $s'$ act equally on $X$; two
  equal-acting $\mathrm{of}$s coincide by faithfulness of $(X,M)$, two
  equal-acting resets have equal targets (states are nonempty by
  Lemma~\ref{lem:nonempty-of-nonunit}).
\end{proof}

\begin{theorem}[{[DKS] Theorem 3.1}]\label{thm:decomposition}
  \lean{KRTheory.TransMon.decomposition}\leanok
  \uses{def:decomp-covers,lem:decomp-tag,lem:decomp-cover-unique,
        lem:bar-generated,
        lem:nonempty-of-nonunit,def:sdiv,def:localdiv}
  If $(X, M)$ is faithful, $M = \langle N \cup \{c\}\rangle$ with
  $N \le M$ a submonoid and $c$ not a unit, then
  \[
  \overline{(X,M)} \;\prec\;
  \overline{(Xc, M_c)} \wr \overline{(X \mathbin{\dot\cup} N, N)}.
  \]
\end{theorem}
\begin{proof}
  \leanok
  The covering submonoid is the closure of the covers; the value map
  $\psi$ sends $w$ to the unique element covered by $w$ whose constructor class
  matches $w$'s tag (existence by induction over the closure — covers
  multiply, and ShapeOK persists by Lemma~\ref{lem:decomp-tag};
  uniqueness by Lemma~\ref{lem:decomp-cover-unique}). $\psi$ is a
  homomorphism: $ww'$ covers $\psi(w)\psi(w')$ and its constructor
  class is the product of the tags, which is $ww'$'s tag. $\psi$ is
  onto: the generators of $\bar M$ (Lemma~\ref{lem:bar-generated}
  applied to $M = \langle N \cup \{c\}\rangle$) are the $\psi$-images of
  the covers. $\varphi$ is onto via
  $x \mapsto (p_0, \mathrm{inl}\ x)$ — some $p_0 \in Xc$ exists by
  Lemma~\ref{lem:nonempty-of-nonunit}. Equivariance is the covering
  property of $\psi(w)$.
\end{proof}
